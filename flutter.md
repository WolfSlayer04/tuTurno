  (function initTurnHistoryInteractions() {
    const filterButtons = document.querySelectorAll('.filter-btn');
    const historyItems = document.querySelectorAll('.history-item');
    const emptyState = document.getElementById('empty-state');
    const searchToggleBtn = document.getElementById('search-toggle-btn');
    const searchInputBox = document.getElementById('search-input-box');
    const searchInput = document.getElementById('restaurant-search');
    const clearSearchBtn = document.getElementById('clear-search-btn');
    const resetFilterBtn = document.getElementById('reset-filter-btn');

    let currentFilter = 'all';
    let searchQuery = '';

    function applyFilters() {
      let visibleCount = 0;

      historyItems.forEach(item => {
        const itemStatus = item.getAttribute('data-status');
        const itemName = item.getAttribute('data-name') || '';

        const matchesStatus = (currentFilter === 'all') || (itemStatus === currentFilter);
        const matchesSearch = !searchQuery || itemName.toLowerCase().includes(searchQuery.toLowerCase());

        if (matchesStatus && matchesSearch) {
          item.classList.remove('hidden');
          visibleCount++;
        } else {
          item.classList.add('hidden');
        }
      });

      if (visibleCount === 0) {
        emptyState.classList.remove('hidden');
      } else {
        emptyState.classList.add('hidden');
      }
    }

    filterButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        filterButtons.forEach(b => {
          b.classList.remove('bg-primary', 'text-on-primary', 'active-filter', 'shadow-sm');
          b.classList.add('bg-surface-container', 'text-on-surface');
        });

        btn.classList.add('bg-primary', 'text-on-primary', 'active-filter', 'shadow-sm');
        btn.classList.remove('bg-surface-container', 'text-on-surface');

        currentFilter = btn.getAttribute('data-filter');
        applyFilters();
      });
    });

    if (searchToggleBtn && searchInputBox) {
      searchToggleBtn.addEventListener('click', () => {
        const isHidden = searchInputBox.classList.contains('hidden');
        if (isHidden) {
          searchInputBox.classList.remove('hidden');
          searchInput.focus();
        } else {
          searchInputBox.classList.add('hidden');
          searchQuery = '';
          searchInput.value = '';
          applyFilters();
        }
      });
    }

    if (searchInput) {
      searchInput.addEventListener('input', (e) => {
        searchQuery = e.target.value.trim();
        applyFilters();
      });
    }

    if (clearSearchBtn) {
      clearSearchBtn.addEventListener('click', () => {
        searchInput.value = '';
        searchQuery = '';
        applyFilters();
      });
    }

    if (resetFilterBtn) {
      resetFilterBtn.addEventListener('click', () => {
        const allBtn = document.querySelector('.filter-btn[data-filter="all"]');
        if (allBtn) allBtn.click();
        if (searchInput) {
          searchInput.value = '';
          searchQuery = '';
        }
        applyFilters();
      });
    }
  })();
